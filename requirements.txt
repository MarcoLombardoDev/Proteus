# Proteus - runtime dependencies

# pillow >= 9.1 is mandatory: Image.Resampling was introduced there.

# ttkbootstrap is optional (the app works with the standard ttk themes) but
# improves the look. 2.x dropped `bootstyle` support on standard ttk widgets;
# the app no longer uses it, so either major version is fine.
ttkbootstrap>=1.10

# Needed to find and replace images inside PDF files. BSD-3-Clause, which is
# what makes it usable here: COMMERCIAL-LICENSE.md promises buyers that no
# dependency imposes copyleft, so PyMuPDF (AGPL or a paid Artifex licence) is
# ruled out however convenient it would be.
#
# Optional at runtime: without it the application starts and does everything
# else, and the PDF option reports that the package is missing instead of
# failing. It is listed here because PDF support is part of the product.
pypdf>=4.0

# Needed only to build the standalone executable (build.py).
pyinstaller>=5.13
